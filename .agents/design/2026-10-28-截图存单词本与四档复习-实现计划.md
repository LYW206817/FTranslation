- **不写 MVVM**：`x:Name` + code-behind 事件处理，与现有四个窗口一致。

```csharp
namespace FastTranslation.Core.Notes;

/// <summary>
/// How well the user recalled a card. Order matters: the higher the value, the further out the card
/// is scheduled. The wire names are the API contract and must not change.
/// </summary>
public enum ReviewGrade
{
    Forgot = 0,
    Fuzzy = 1,
    Familiar = 2,
    Known = 3,
}

/// <summary>
/// A fixed ladder, not SM-2. Design §5: the product asks for "classify it and bring it back later",
/// which two counters and one lookup table answer exactly. Pure functions only — no clock, no IO —
/// so the 32-cell transition table is testable without a fixture.
/// </summary>
public static class ReviewSchedule
{
    public const int MaxStage = 7;

    /// <summary>Seconds to wait after grading, indexed by the resulting stage. Index 0 means "never graded".</summary>
    public static readonly IReadOnlyList<long> IntervalsSeconds = new long[]
    {
        0L, 600L, 86_400L, 259_200L, 604_800L, 1_209_600L, 2_419_200L, 4_838_400L,
    };

    private static readonly string[] Labels =
    [
        "未排期", "10 分钟后", "1 天后", "3 天后", "7 天后", "14 天后", "28 天后", "56 天后",
    ];

    /// <summary>
    /// Four grades map onto four rungs. 记住 jumps to 7 days on the first try (the user already knows
    /// it, there is no reason to show it again today) and then walks up one rung per success up to 56
    /// days. Every downgrade has a floor of stage 1: stage 0 stays reserved for "never graded", so a
    /// forgotten card is never demoted back into the new-card pile (D10).
    /// </summary>
    public static int NextStage(int stage, ReviewGrade grade)
    {
        var current = Math.Clamp(stage, 0, MaxStage);
        return grade switch
        {
            ReviewGrade.Known => current >= 4 ? Math.Min(MaxStage, current + 1) : 4,
            ReviewGrade.Familiar => 3,
            ReviewGrade.Fuzzy => 2,
            _ => 1,
        };
    }

    /// <summary>Resulting stage plus the absolute due time in unix seconds.</summary>
    public static (int Stage, long DueAt) Schedule(int currentStage, ReviewGrade grade, long nowUnix)
    {
        var stage = NextStage(currentStage, grade);
        return (stage, nowUnix + IntervalsSeconds[stage]);
    }

    public static bool TryParseGrade(string? wire, out ReviewGrade grade)
    {
        switch ((wire ?? string.Empty).Trim().ToLowerInvariant())
        {
            case "forgot": grade = ReviewGrade.Forgot; return true;
            case "fuzzy": grade = ReviewGrade.Fuzzy; return true;
            case "familiar": grade = ReviewGrade.Familiar; return true;
            case "known": grade = ReviewGrade.Known; return true;
            default: grade = ReviewGrade.Forgot; return false;
        }
    }

    public static string ToWire(ReviewGrade grade) => grade switch
    {
        ReviewGrade.Forgot => "forgot",
        ReviewGrade.Fuzzy => "fuzzy",
        ReviewGrade.Familiar => "familiar",
        _ => "known",
    };

    /// <summary>The little caption under each grade button ("7 天后"), derived locally so it shows before any request.</summary>
    public static string IntervalLabel(int stage) => Labels[Math.Clamp(stage, 0, MaxStage)];

    /// <summary>Display order of the four buttons, left to right.</summary>
    public static readonly IReadOnlyList<ReviewGrade> AllGrades = new[]
    {
        ReviewGrade.Forgot, ReviewGrade.Fuzzy, ReviewGrade.Familiar, ReviewGrade.Known,
    };

    public static string Display(ReviewGrade grade) => grade switch
    {
        ReviewGrade.Forgot => "忘记",
        ReviewGrade.Fuzzy => "模糊",
        ReviewGrade.Familiar => "熟悉",
        _ => "记住",
    };
}
```

```csharp
public sealed record Note(
    Guid Id,
    NoteKind Kind,
    string SourceText,
    string TargetText,
    string SourceLang,
    string TargetLang,
    string Model,
    string OriginApp,
    string Annotation,
    long CreatedAt,
    long UpdatedAt,
    int Stage = 0,
    long DueAt = 0,
    string LastGrade = "",
    int ReviewCount = 0,
    int Lapses = 0,
    long ReviewedAt = 0)
{
    public string KindDisplay => NoteKinds.Display(Kind);

    /// <summary>"忘记 / 模糊 / 熟悉 / 记住 / 未评" — the status line in the detail pane.</summary>
    public string GradeDisplay => LastGrade switch
    {
        "forgot" => "上次:忘记",
        "fuzzy" => "上次:模糊",
        "familiar" => "上次:熟悉",
        "known" => "上次:记住",
        _ => "还没评过",
    };

    /// <summary>"3 天后" / "今天到期" / "未排期" — third line of a list row.</summary>
    public string DueLine
    {
        get
        {
            if (Stage == 0) return "未排期";
            var today = DateTimeOffset.Now.Date;
            var due = DateTimeOffset.FromUnixTimeSeconds(DueAt).ToLocalTime().Date;
            var days = (due - today).Days;
            return days switch
            {
                < 0 => "今天到期",
                0 => "今天到期",
                1 => "明天",
                _ => days + " 天后",
            };
        }
    }

    /// <summary>"chrome · 10-28" — the provenance line under a card, never the window title.</summary>
    public string OriginLine
    {
        get
        {
            var when = DateTimeOffset.FromUnixTimeSeconds(CreatedAt).ToLocalTime().ToString("MM-dd");
            return string.IsNullOrWhiteSpace(OriginApp) ? when : OriginApp + " · " + when;
        }
    }
}
```

```csharp
    private const int DefaultNewLimit = 20;
    private const int DefaultQueueLimit = 50;

    /// <summary>
    /// Today's queue. <paramref name="dayStart"/> is the caller's local midnight in unix seconds —
    /// the server never guesses a time zone (design §5.4).
    /// </summary>
    private static IResult ReviewQueue(
        [FromQuery] long? dayStart,
        [FromQuery] int? newLimit,
        [FromQuery] int? limit,
        ClaimsPrincipal principal,
        NoteStore store,
        AccountStore accounts,
        ServerDatabase database)
    {
        var userId = principal.GetUserId();
        var user = accounts.FindUserById(userId);
        if (user is null || user.Disabled)
        {
            return Results.Json(new ErrorResponse("account_disabled"), statusCode: StatusCodes.Status403Forbidden);
        }

        if (!Entitlement.AllowsNotebook(user.Plan, user.PlanExpiresAt, database.NowUnix))
        {
            return Results.Json(new ErrorResponse("subscription_required"), statusCode: StatusCodes.Status403Forbidden);
        }

        var start = dayStart is > 0 ? dayStart.Value : StartOfTodayUtc(database.NowUnix);
        var wantedNew = Math.Clamp(newLimit ?? DefaultNewLimit, 0, 100);
        var take = Math.Clamp(limit ?? DefaultQueueLimit, 1, 200);

        var due = store.ListDue(userId, database.NowUnix, take);
        var fresh = store.ListFresh(userId, start, Math.Max(1, wantedNew));
        var alreadyQueued = store.CountFreshSince(userId, start);
        var quota = Math.Max(0, wantedNew - alreadyQueued);

        return Results.Ok(new ReviewQueueResponse(
            due.Select(ToDto).ToList(),
            fresh.Items.Select(ToDto).ToList(),
            due.Count,
            quota,
            take,
            wantedNew));
    }

    private static IResult Grade(
        Guid id,
        [FromBody] GradeNoteRequest request,
        ClaimsPrincipal principal,
        NoteStore store,
        AccountStore accounts,
        ServerDatabase database)
    {
        var userId = principal.GetUserId();
        var user = accounts.FindUserById(userId);
        if (user is null || user.Disabled)
        {
            return Results.Json(new ErrorResponse("account_disabled"), statusCode: StatusCodes.Status403Forbidden);
        }

        if (!Entitlement.AllowsNotebook(user.Plan, user.PlanExpiresAt, database.NowUnix))
        {
            return Results.Json(new ErrorResponse("subscription_required"), statusCode: StatusCodes.Status403Forbidden);
        }

        if (!ReviewSchedule.TryParseGrade(request.Grade, out var grade))
        {
            return Results.BadRequest(new ErrorResponse("grade_invalid"));
        }

        var existing = store.FindById(userId, id);
        if (existing is null)
        {
            return Results.NotFound(new ErrorResponse("note_missing"));
        }

        var now = database.NowUnix;
        var (stage, dueAt) = ReviewSchedule.Schedule(existing.Stage, grade, now);
        var updated = store.SetReview(userId, id, stage, dueAt, ReviewSchedule.ToWire(grade), now);
        return updated is null
            ? Results.NotFound(new ErrorResponse("note_missing"))
            : Results.Ok(ToDto(updated));
    }

    /// <summary>Fallback when the client omits dayStart: UTC midnight. Clients always send their own local midnight.</summary>
    private static long StartOfTodayUtc(long nowUnix) => nowUnix - (nowUnix % 86_400L);
```

```csharp
    private static async Task<IResult> Ocr(
        HttpContext context,
        ClaimsPrincipal principal,
        DashScopeGateway upstream,
        UsageStore store,
        AccountStore accounts,
        GatewayOptions gateway,
        RateLimiter limiter)
    {
        if (!gateway.IsConfigured) return Fail(StatusCodes.Status503ServiceUnavailable, "gateway_disabled");

        var body = await ReadBodyAsync(context.Request).ConfigureAwait(false);
        if (string.IsNullOrWhiteSpace(body)) return Fail(StatusCodes.Status400BadRequest, "body_invalid");

        // One OCR is priced like one image translation for quota purposes: both take a picture in.
        var quotaChars = gateway.ImageQuotaChars > 0 ? gateway.ImageQuotaChars : 1_000;
        if (Intent(context, principal, body, quotaChars, gateway, store, accounts, limiter) is { } refusal)
        {
            return refusal;
        }

        var response = await upstream.SendOcrAsync(body, context.RequestAborted).ConfigureAwait(false);
        if (!response.Succeeded)
        {
            // Same rule as /chat/completions: an upstream failure is forwarded verbatim and is not billed.
            return Results.Text(response.Body, statusCode: response.StatusCode, contentType: "application/json");
        }

        var model = gateway.OcrModel;
        _ = DashScopeGateway.TryReadUsage(response.Body, out var tokensIn, out var tokensOut);
        if (tokensIn <= 0) tokensIn = quotaChars;
        if (tokensOut <= 0) tokensOut = 200; // 一块截图的一行字量级，只用于成本统计

        store.RecordUsage(
            principal.GetUserId().ToString("D"),
            model,
            quotaChars,
            Pricing.TextCostMicros(model, tokensIn, tokensOut),
            tokensIn,
            tokensOut,
            GatewayEndpoints.Device(context, principal));

        return Results.Text(response.Body, contentType: "application/json");
    }
```

> `Intent(...)` 现在是 `private static IResult? Intent(HttpContext, ClaimsPrincipal, string body, ...)` 里内部调 `ReadChatShape`。OCR 传进来的 `body` 是图片 body，`Intent` 必须**不读 chars**，只做 rate/budget/account/quota 四道门禁且用外部传入的 `quotaChars`。检查 `Intent` 当前签名：它已经接收 `int chars`（在 `ChatCompletions` 里由 `ReadChatShape` 算出后传入），所以 OCR 直接把 `quotaChars` 传进去即可，**不需要改 `Intent`**。

```csharp
namespace FastTranslation.Core;

/// <summary>The screenshot to run OCR on. Only the PNG is needed; languages are auto-detected upstream.</summary>
public sealed record ImageOcrRequest(byte[] PngBytes);

/// <summary>Extracted text, verbatim. Unlike <see cref="ImageTranslationResult"/> this carries no image.</summary>
public sealed record ImageOcrResult(string Text, long ElapsedMs, string RemoteUrl)
{
    public TranslationError? Error { get; init; }
    public string? Notice { get; init; }
    public bool Succeeded => Error is null && !string.IsNullOrWhiteSpace(Text);

    public static ImageOcrResult Failed(TranslationError error, long elapsedMs) =>
        new(string.Empty, elapsedMs, string.Empty) { Error = error };
}

public interface IImageOcrProvider : IDisposable
{
    Task<ImageOcrResult> RecognizeAsync(ImageOcrRequest request, CancellationToken cancellationToken);
}
```

```csharp
    /// <summary>成功响应 output.choices[0].message.content[0].text；任何一层缺失都返回 false。</summary>
    internal static bool TryReadText(string json, out string text)
    {
        text = string.Empty;
        if (string.IsNullOrWhiteSpace(json)) return false;
        try
        {
            using var doc = JsonDocument.Parse(json);
            if (doc.RootElement.ValueKind != JsonValueKind.Object) return false;
            if (!doc.RootElement.TryGetProperty("output", out var output) || output.ValueKind != JsonValueKind.Object) return false;
            if (!output.TryGetProperty("choices", out var choices) || choices.ValueKind != JsonValueKind.Array) return false;
            if (choices.GetArrayLength() == 0) return false;
            var first = choices[0];
            if (first.ValueKind != JsonValueKind.Object) return false;
            if (!first.TryGetProperty("message", out var message) || message.ValueKind != JsonValueKind.Object) return false;
            if (!message.TryGetProperty("content", out var content)) return false;

            if (content.ValueKind == JsonValueKind.Array)
            {
                foreach (var part in content.EnumerateArray())
                {
                    if (part.ValueKind == JsonValueKind.Object
                        && part.TryGetProperty("text", out var node)
                        && node.ValueKind == JsonValueKind.String)
                    {
                        text = node.GetString() ?? string.Empty;
                        return !string.IsNullOrWhiteSpace(text);
                    }
                }
                return false;
            }

            if (content.ValueKind == JsonValueKind.String)
            {
                text = content.GetString() ?? string.Empty;
                return !string.IsNullOrWhiteSpace(text);
            }
            return false;
        }
        catch (JsonException)
        {
            return false;
        }
    }
```

- [ ] **Step 3: `GatewayOcrProvider.cs`（照抄 `GatewayImageTranslationProvider`）**

```csharp
/// <summary>
/// OCR 的组合 provider，与 <see cref="GatewayImageTranslationProvider"/> 同构：网关不接收
/// （没登录，回调返回 null）或额度用完时，安静改用用户自己的 Key；两路都失败时报自有 Key 的错。
/// </summary>
public sealed class GatewayOcrProvider : IImageOcrProvider
{
    private readonly Func<ImageOcrRequest, CancellationToken, Task<ImageOcrResult?>> _gateway;
    private readonly IImageOcrProvider _own;

    public GatewayOcrProvider(
        Func<ImageOcrRequest, CancellationToken, Task<ImageOcrResult?>> gateway,
        IImageOcrProvider own)
    {
        _gateway = gateway;
        _own = own;
    }

    public async Task<ImageOcrResult> RecognizeAsync(ImageOcrRequest request, CancellationToken cancellationToken)
    {
        var gateway = await _gateway(request, cancellationToken).ConfigureAwait(false);
        if (gateway is null) return await _own.RecognizeAsync(request, cancellationToken).ConfigureAwait(false);
        if (gateway.Succeeded) return gateway;
        if (gateway.Error is { Code: TranslationErrorCode.Cancelled or TranslationErrorCode.ContentTooLong })
        {
            return gateway;
        }

        var own = await _own.RecognizeAsync(request, cancellationToken).ConfigureAwait(false);
        return own.Succeeded ? own with { Notice = Notice(gateway.Error!) } : own;
    }

    private static string? Notice(TranslationError error) => error.Code switch
    {
        TranslationErrorCode.QuotaExhausted => error.Message + "，这次用了你自己的 Key",
        TranslationErrorCode.RateLimited => "免费额度正在排队，这次用了你自己的 Key",
        _ => null,
    };

    public void Dispose() => _own.Dispose();
}
```

```csharp
    private async void Grade_Click(object sender, RoutedEventArgs e)
    {
        if (sender is not Button { Tag: string wire }) return;
```

```csharp
    }

    private async Task GradeAsync(string wire)
    {
        if (!_revealed || _grading || Current is null) return;
        if (!ReviewSchedule.TryParseGrade(wire, out var grade)) return;

        _grading = true;
        SetGradeEnabled(false);
        try
        {
            var result = await _account.GradeNoteAsync(Current.Id, grade, CancellationToken.None);
            if (!result.Succeeded)
            {
                // 卡片留在原张，按钮恢复可点（设计 §9）
                StatusText.Text = result.Error!.Display;
                StatusText.Visibility = Visibility.Visible;
                SetGradeEnabled(true);
                return;
            }
            _index++;
            Render();                   // 队列评完时 Render 自己会走空态
        }
        finally
        {
            _grading = false;
        }
    }
```

> **`SetGradeEnabled` 在 `finally` 里不要再打开**：`Render()` 已经把新卡片的按钮置灰了，`finally` 打开会把未揭示的卡片变成可点。上面写法就是对的——成功路径靠 `Render()` 重置，失败路径自己打开。

```csharp
            return;

                case Key.D2 or Key.NumPad2: _ = GradeAsync("fuzzy"); e.Handled = true; return;
                case Key.D3 or Key.NumPad3: _ = GradeAsync("familiar"); e.Handled = true; return;
                case Key.D4 or Key.NumPad4: _ = GradeAsync("known"); e.Handled = true; return;

    }
```
